# Core dependencies for the YouTube learning tools.
# Pinned to known-good major versions for reproducibility.
#
# SYSTEM DEPENDENCY (NOT installable via pip): ffmpeg
#   yt-dlp needs the ffmpeg binary to convert/merge audio (e.g. to real .mp3).
#   Install it separately, then confirm with `ffmpeg -version`:
#     conda:  conda install -c conda-forge ffmpeg -y   (installs into this env)
#     winget: winget install ffmpeg                    (system-wide)
#     choco:  choco install ffmpeg -y                  (admin, system-wide)
#     macOS:  brew install ffmpeg
#     Debian/Ubuntu: sudo apt install ffmpeg

# Fetch video transcripts/subtitles (fixed typo: was "youtoube-transcript-api")
youtube-transcript-api>=0.6,<1

# Audio/video download + listing + subtitles. This is the core downloader used
# by all active tools. The [default,curl-cffi] extra pulls in curl_cffi, which
# yt-dlp uses for browser "impersonation" (a real TLS fingerprint). Without it,
# YouTube now blocks subtitle/format fetches ("Sign in to confirm you're not a
# bot", empty subtitle responses). This is REQUIRED for downloads/transcripts.
yt-dlp[default,curl-cffi]>=2024.4.9

# --- Legacy, only for archived/legacy scripts (not the active workflow) ---
# pytube: still imported by code/youtube_download.py (fragile; superseded by
#   yt-dlp). google-api-python-client: only used by the retired
#   ignore/get_my_playlists.py. scrapetube was dropped entirely (broken since
#   2025, no active code uses it). Keep pytube only while youtube_download.py
#   remains; the others can go once the notebook is migrated to yt-dlp.
pytube>=15.0,<16
google-api-python-client>=2.100,<3

# curl_cffi is also listed explicitly so `import curl_cffi` is guaranteed
# present even if the extra name changes across yt-dlp versions.
curl_cffi>=0.7

# Data wrangling for metadata tables
pandas>=2.0,<3

# Jupyter kernel, so the notebooks can run in this environment
ipykernel>=6.29,<7

# Speech-to-text (Phase C): transcribe audio whose YouTube captions are
# disabled. faster-whisper (CTranslate2 backend) is lighter/faster than
# openai-whisper, CPU-friendly (int8), and can also translate. It downloads
# the model weights from Hugging Face on first use, and needs ffmpeg (already
# a system dependency above).
faster-whisper>=1.0
